\begin{frame}{Kausalitas dan Timestamp}

  \draw[flow] (b)--(d); \draw[flow] (c)--(d);
\end{tikzpicture}
\medskip

\small
\begin{itemize}
  \item Timestamp memberi satu urutan total, padahal jam antarlayanan dapat berbeda.
  \item Kausalitas (\emph{happened-before}, Lamport 1978) hanya menuntut sebagian pasangan berurutan: pengiriman menunggu pembayaran dan reservasi.
\end{itemize}
\note{Pada alur pesanan, pembayaran dan reservasi berjalan konkuren dan boleh selesai dalam urutan apa pun; yang wajib hanyalah keduanya mendahului pengiriman. Mengurutkan dengan timestamp memaksa satu urutan total dan bisa keliru karena jam tiap layanan berbeda. Relasi happened-before dari Lamport hanya mengurutkan pasangan yang memang punya bukti kausal.}
\end{frame}

\begin{frame}{Telemetri dan Bukti Bisnis}
\small
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\textbf{Titik peristiwa}
\medskip

\texttt{payment.requested} (span dimulai) $\neq$ \texttt{payment.succeeded} (aksi berhasil).
\medskip

OpenTelemetry memberi struktur (trace, span, atribut), bukan makna bisnis.
\end{column}
\begin{column}{0.47\textwidth}
\textbf{Bukti tidak lengkap}
\medskip

Peristiwa yang tidak ada di ekspor bisa berarti tidak terjadi \emph{atau} tidak terekam (sampling, batas perekaman).
\medskip

Menyamakan keduanya menimbulkan alarm tanpa dasar.
\end{column}
\end{columns}
\vfill
\begin{center}
\normalsize\textbf{Perlu kontrak observasi dan status \emph{tidak dapat disimpulkan}.}
\end{center}
\vspace{0.4cm}
\note{Masalah kedua: data teknis tidak sama dengan peristiwa bisnis. Awal span pembayaran belum berarti pembayaran berhasil, jadi titik pencatatan harus ditetapkan. Masalah ketiga: peristiwa yang tidak terlihat di ekspor bisa saja terjadi tetapi tidak terekam. Karena itu kami butuh kontrak observasi dan keputusan ketiga, yaitu tidak dapat disimpulkan.}
\end{frame}

  \item awal span induk ke awal span anak;

Atribut porv bukan atribut standar OpenTelemetry dan kami laporkan sebagai bagian kontrak.}
\end{frame}

\begin{frame}{Pencarian Urutan}
\smallitems\small
Untuk graf $G=(V,D)$ dan model DFA $P$ ($\operatorname{TopSort}(G)$: himpunan urutan topologis):
\abovedisplayskip=3pt\belowdisplayskip=3pt
\[
  \operatorname{kompatibel}(G,P)
  \iff \exists\pi\in\operatorname{TopSort}(G):
  \lambda(\pi)\in L(P).
\]
\textbf{Kompatibel $\not\Rightarrow$ aplikasi benar.} Contoh model ketat hanya menerima $abcd$; satu urutan diterima sudah cukup.
\begin{columns}[T]
\begin{column}{0.40\textwidth}
\centering
\begin{tikzpicture}[level distance=0.47cm,
  level 2/.style={sibling distance=3.6cm}, every node/.style={font=\small},
  cfg/.append style={font=\small,inner sep=3.5pt},
  acc/.append style={font=\small,inner sep=3.5pt},
  rej/.append style={font=\small,inner sep=3.5pt}]
\node[cfg]{$\varnothing$}
  child { node[cfg]{$a$}
    child { node[cfg]{$ab$}
      child { node[cfg]{$abc$}
        child { node[acc]{$abcd$ $\checkmark$} } } }
    child { node[rej,label={[font=\scriptsize,text=gray,align=center,yshift=-3pt]below:{DFA tidak punya\\transisi $c$}}]{$ac$ $\times$} } };
\end{tikzpicture}
{\scriptsize
\begin{tabular}{@{}ll@{\quad}ll@{}}
$a$ & pesanan dibuat & $b$ & pembayaran \\
$c$ & reservasi & $d$ & pengiriman
\end{tabular}}
\end{column}
\begin{column}{0.58\textwidth}
\begin{itemize}
  \setlength{\itemsep}{0pt}
  \item \textbf{Langkah:} pilih peristiwa setelah semua pendahulunya. Ikuti transisi DFA dan simpan tiap konfigurasi $(S,q)$ sekali.
  \item \textbf{Biaya:} jumlah konfigurasi $\leq |Q|(n+1)^w$, dengan $w$ lebar poset.
  \item \textbf{Hasil:} saksi $\rightarrow$ kompatibel; pilihan habis $\rightarrow$ tidak kompatibel; batas 10 detik / $10^6$ konfigurasi $\rightarrow$ belum dapat disimpulkan.
\end{itemize}
\end{column}
\end{columns}
\note{Observasi kompatibel jika ada sedikitnya satu urutan topologis yang, setelah dipetakan ke simbol bisnis, diterima DFA. S adalah peristiwa yang sudah diproses dan q keadaan DFA; setiap pasangan disimpan sekali. Di contoh ini model ketat tidak punya transisi c setelah a, jadi cabang ac langsung berhenti; cabang kiri sampai ke abcd yang diterima, jadi itu saksinya. Masalah umumnya NP-complete menurut Amarilli dan Paperman, tetapi jumlah konfigurasi polinomial untuk lebar w tetap, sehingga lebar menjadi faktor eksperimen.}
\end{frame}

Penolakan butuh pencarian tuntas, sehingga batas sumber daya menghasilkan tidak dapat disimpulkan, bukan penolakan.}

\small
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\textwidth}{@{}Xccc@{}}
\toprule
Pembanding & Sisi kausal & Batas jam & Cari alternatif \\
\midrule
Timestamp langsung & Tidak & Tidak & Tidak \\

\medskip

Timestamp diuji dengan pergeseran 0, $\pm100$ ms, $\pm1$ s, dan $\pm5$ menit. Kita ukur ketepatan, cakupan, dan biaya.
\smallskip

\footnotesize Timestamp vs satu urutan topologis: \textbf{manfaat informasi kausal}. Satu urutan topologis vs PORV: \textbf{manfaat pencarian alternatif}.
\note{Semua strategi memakai peristiwa, pemetaan simbol, DFA, dan gerbang bukti yang sama; hanya cara mengurutkannya yang berbeda. Perbandingan timestamp dengan satu urutan topologis mengukur manfaat informasi kausal, sedangkan satu urutan topologis dengan PORV mengukur manfaat pencarian. Bila model menerima semua pertukaran peristiwa tidak terurut, pencarian tidak diharapkan menambah ketepatan.}
\end{frame}

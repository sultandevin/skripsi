\begin{frame}{Satu saksi menerima model, pencarian memeriksa pilihan yang mungkin}
\small
Untuk graf $G=(V,D)$ dan model DFA $P$:
\abovedisplayskip=3pt\belowdisplayskip=3pt
\[
  \operatorname{kompatibel}(G,P)
  \iff \exists\pi\in\operatorname{TopSort}(G):
  \lambda(\pi)\in L(P).
\]
\textbf{Kompatibel $\not\Rightarrow$ aplikasi benar.} Graf mengizinkan kedua urutan. DFA secara terpisah menerima $abcd$ tetapi menolak $acbd$; satu urutan diterima sudah cukup untuk kompatibel.
\begin{columns}[T]
\begin{column}{0.42\textwidth}
\centering
\begin{tikzpicture}[level distance=0.52cm,
  level 2/.style={sibling distance=3.6cm}, every node/.style={font=\small},
  cfg/.append style={font=\small,inner sep=3.5pt},
  acc/.append style={font=\small,inner sep=3.5pt},
  rej/.append style={font=\small,inner sep=3.5pt}]
\node[cfg]{$\varnothing$}
  child { node[cfg]{$a$}
    child { node[cfg]{$ab$}
      child { node[cfg]{$abc$}
        child { node[acc]{$abcd$ $\checkmark$} } } }
    child { node[cfg]{$ac$}
      child { node[rej]{$acb$}
        child { node[rej]{$acbd$ $\times$} } } } };
\end{tikzpicture}
{\scriptsize
\begin{tabular}{@{}ll@{\quad}ll@{}}
$a$ & pesanan dibuat & $b$ & pembayaran \\
$c$ & reservasi & $d$ & pengiriman
\end{tabular}}
\end{column}
\begin{column}{0.56\textwidth}
\begin{itemize}
  \setlength{\itemsep}{0pt}
  \item \textbf{Langkah:} pilih peristiwa setelah semua pendahulunya. Ikuti transisi DFA dan simpan tiap pasangan $(S,q)$ sekali.
  \item \textbf{Biaya dan hasil:} $|Q|(n+1)^w$. Saksi berarti kompatibel. Pilihan habis berarti tidak kompatibel. Batas 10 detik atau $10^6$ konfigurasi berarti belum dapat disimpulkan.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Cadangan: analisis statistik}
\small\setlength{\itemsep}{6pt}
\begin{itemize}
  \item Unit independen: satu graf dasar atau satu eksekusi aplikasi baru. Perubahan jam, kehilangan, dan pengulangan tetap satu kelompok.
  \item Bootstrap berpasangan 10.000 replikasi untuk selisih kesalahan, cakupan, dan median selisih waktu. Interval 95\%.
  \item McNemar eksak pada pasangan keputusan biner untuk label aplikasi, dengan koreksi Holm pada keluarga delapan uji.
  \item FPR/FNR dilaporkan dengan pembilang, penyebut, dan interval binomial eksak pada observasi dasar independen.
  \item Status tidak dapat disimpulkan dilaporkan terpisah. Ketepatan bersyarat selalu disertai cakupan.
\end{itemize}
\end{frame}

\begin{frame}{Cadangan: pemeriksaan prasyarat kausal}
\small\setlength{\itemsep}{6pt}
\begin{itemize}
  \item Pertanyaan: apakah ada lintasan kausal dari prasyarat ke aksi dependen?
  \item Profil dibatasi pada dependensi eksplisit tanpa pengulangan. Oracle lintasan berbeda dari oracle DFA.
  \item Tidak adanya lintasan berarti prasyarat tidak didukung bukti. Itu belum tentu kesalahan aplikasi.
  \item Pada S7, hasil ini melengkapi interpretasi penerimaan eksistensial.
  \item Kalau dipakai sebagai alarm, TP/FP/FN/TN dihitung ulang terhadap fakta bisnis independen.
\end{itemize}
\end{frame}
